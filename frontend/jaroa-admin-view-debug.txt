     1	<?php
     2	
     3	namespace App;
     4	
     5	use App\Templates\TemplateManager;
     6	use RuntimeException;
     7	
     8	class View
     9	{
    10	    private static array $shared = [];
    11	
    12	    private static ?TemplateManager $templateManager = null;
    13	
    14	    /**
    15	     * Share data with all views.
    16	     */
    17	    public static function share(array $data): void
    18	    {
    19	        self::$shared = array_merge(
    20	            self::$shared,
    21	            $data
    22	        );
    23	    }
    24	
    25	    /**
    26	     * Set the template manager.
    27	     */
    28	    public static function setTemplateManager(
    29	        TemplateManager $templateManager
    30	    ): void {
    31	        self::$templateManager = $templateManager;
    32	    }
    33	
    34	    /**
    35	     * Render a normal application view.
    36	     */
    37	    public static function render(
    38	        string $view,
    39	        array $data = []
    40	    ): void {
    41	        self::renderFile(
    42	            self::viewPath($view),
    43	            $data
    44	        );
    45	    }
    46	
    47	    /**
    48	     * Render the currently active template.
    49	     */
    50	    public static function renderTemplate(
    51	        array $data = []
    52	    ): void {
    53	        if (null === self::$templateManager) {
    54	            throw new RuntimeException(
    55	                'Template manager has not been configured.'
    56	            );
    57	        }
    58	
    59	        $template = self::$templateManager->active();
    60	
    61	        if (null === $template) {
    62	            throw new RuntimeException(
    63	                'No active template.'
    64	            );
    65	        }
    66	
    67	        self::renderFile(
    68	            self::templatePath($template),
    69	            $data
    70	        );
    71	    }
    72	
    73	    /**
    74	     * Render an installed template by its slug.
    75	     */
    76	    public static function renderTemplateBySlug(
    77	        string $slug,
    78	        array $data = []
    79	    ): void {
    80	        if (null === self::$templateManager) {
    81	            throw new RuntimeException(
    82	                'Template manager has not been configured.'
    83	            );
    84	        }
    85	
    86	        if (!self::$templateManager->isInstalled($slug)) {
    87	            throw new RuntimeException(
    88	                "Template is not installed: {$slug}"
    89	            );
    90	        }
    91	
    92	        self::renderFile(
    93	            self::templatePath($slug),
    94	            $data
    95	        );
    96	    }
    97	
    98	    /**
    99	     * Render a view file through the application layout.
   100	     */
   101	    private static function renderFile(
   102	        string $viewPath,
   103	        array $data
   104	    ): void {
   105	        extract(
   106	            array_merge(
   107	                self::$shared,
   108	                $data
   109	            ),
   110	            EXTR_SKIP
   111	        );
   112	
   113	        ob_start();
   114	
   115	        require $viewPath;
   116	
   117	        $content = ob_get_clean();
   118	
   119	        $layoutPath =
   120	            __DIR__ . '/../views/layouts/app.php';
   121	
   122	        if (!is_file($layoutPath)) {
   123	            throw new RuntimeException(
   124	                'Application layout not found.'
   125	            );
   126	        }
   127	
   128	        require $layoutPath;
   129	    }
   130	
   131	    /**
   132	     * Resolve a normal application view path.
   133	     */
   134	    private static function viewPath(
   135	        string $view
   136	    ): string {
   137	        $viewPath =
   138	            __DIR__ .
   139	            '/../views/' .
   140	            $view .
   141	            '.php';
   142	
   143	        if (!is_file($viewPath)) {
   144	            throw new RuntimeException(
   145	                "View not found: {$view}"
   146	            );
   147	        }
   148	
   149	        return $viewPath;
   150	    }
   151	
   152	    /**
   153	     * Resolve an installed theme template path.
   154	     */
   155	    private static function templatePath(
   156	        string $slug
   157	    ): string {
   158	        if (null === self::$templateManager) {
   159	            throw new RuntimeException(
   160	                'Template manager has not been configured.'
   161	            );
   162	        }
   163	
   164	        $themePath =
   165	            __DIR__ .
   166	            '/../site/themes/' .
   167	            $slug .
   168	            '/template.php';
   169	
   170	        if (!is_file($themePath)) {
   171	            throw new RuntimeException(
   172	                "Installed template not found: {$slug}"
   173	            );
   174	        }
   175	
   176	        return $themePath;
   177	    }
   178	}
